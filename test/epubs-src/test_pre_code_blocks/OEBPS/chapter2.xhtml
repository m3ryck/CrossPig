PASS when indentation, blank lines, and line breaks survive, while long lines still wrap inside the viewport.

JavaScript

function buildPages(chapter) {
    const blocks = parseHtml(chapter);

    for (const block of blocks) {
        if (block.kind === "pre") {
            keepWhitespace(block.text);
        }
    }

	return blocks.length;
}

C++

bool preserveWhitespace(const char* text) {
    if (text == nullptr) {
        return false;
    }

    // Spaces below should stay aligned.
    const int columns[] = { 1,  20,   300 };
    return columns[0] < columns[2];
}
